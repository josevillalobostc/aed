struct LRUCache(int n):
  Dict dict<key, ptr>
  DoublyLinkedList list<key,value>
  capacity = n
  size = 0

  move_to_front(ptr):
    if(ptr == list.head) 
      return
    ptr.prev.next = ptr.next
    if(ptr == list.tail):
      list.tail = ptr.prev
    else:
      ptr.next.prev = ptr.prev
    list.head.prev = ptr
    ptr.next = list.head
    ptr.prev = nullptr
    list.head = ptr

  put(key, value):
    ptr = dict[key]
    if(ptr == nullptr):
      if size == capacity:
        dict.erase(list.tail.key)
        list.pop_back
        size--
      list.push_front(key, value)
      dict[key] = list.head
      size++
    else:
      move_to_front(ptr)
      ptr.value = value

  get(key):
    if(not dict.has_key(key)):
      return -1
    ptr = dict[key]
    move_to_front(ptr)
    return ptr.value

validarBST(root):
  validar(root, -inf, +inf)

validar(root, left, right):
  if(root == nullptr):
    return true
  if(left >= root.value):
    return false
  if(right <= root.value):
    return false
  return validar(root.left, left, root.value) and validar(root.right, root.value, right)

merge_lists(vector<list> lists, k):
  min_heap heep
  list res
  bool finished = false
  while(!finished):
    finished = true
    for(list in lists):
      if(list.head != nullptr):
        finished = false
        heep.insert(list.head.value)
        heep.pop_front()
  while(!heep.empty()):
    res.push_back(heep.pop())
  return res
